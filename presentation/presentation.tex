% ---------------------------------------------------------------------------
\begin{frame}{The rules: closed book}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Allowed}
        \begin{itemize}
          \item[\ok] Open-weight models, $\le$\,32B published parameters
          \item[\ok] In-context examples from the training split
          \item[\ok] Multi-step inference, repeated sampling
          \item[\ok] Non-neural post-processing
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Forbidden}
        \begin{itemize}
          \item[\no] Web, retrieval index, external corpus, knowledge base
          \item[\no] Fine-tuning
          \item[\no] Combining models whose parameters sum above 32B
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{5mm}
  Retrieval would supply exactly the long-tail facts a model's memory covers
  worst. Without it, \alert{prompt diversity and repeated sampling} are the
  only sources of variation left.
\end{frame}

\begin{frame}{Step 1 --- Say who the subject is before answering}
  Many subjects are ambiguous names. One cheap call (256 tokens, low
  temperature) asks the model to describe the exact entity first; that
  description is handed to every later pass as \emph{uncertain} context.

  \vspace{2mm}
  \begin{center}
  \begin{tikzpicture}[node distance=3mm]
    \node[subj] (q) {``University Stadium in Georgia'', \rel{hasCapacity}};
    \node[stage, right=7mm of q, text width=56mm, align=left] (a)
      {Home field of the University of Georgia Bulldogs in Athens, Georgia ---
       officially \textbf{Sanford Stadium}.};
    \draw[arr] (q) -- (a);
    \node[ann, above=1mm of a] {anchor};
  \end{tikzpicture}
  \end{center}

  \vspace{1mm}
  \begin{itemize}
    \item Resolves namesakes, scope, and ``is this company independently listed?''
    \item It can also propagate a wrong guess: \emph{Ayon Island} (East Siberian
      Sea) was anchored to the Philippines and every sample followed.
    \item \alert{Not ablated}: present in every run, so its net value is unmeasured.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Step 2 --- Ask many times, each time differently}
  Each pass uses a different seed, a different \emph{recall focus}, and a
  different rotating set of training examples. Passes run concurrently.

  \vspace{2mm}
  \begin{center}
  \resizebox{0.95\textwidth}{!}{%
  \begin{tikzpicture}[node distance=2.5mm]
    \node[subj] (q) {Lo\v{s}inj, \rel{hasArea}};
    \node[sample, right=8mm of q] (p1) {128};
    \node[sample, right=of p1] (p2) {126};
    \node[sample, right=of p2] (p3) {128};
    \node[sample, right=of p3] (p4) {129};
    \node[sample, right=of p4] (p5) {128};
    \node[sample, right=of p5] (p6) {128};
    \node[sample, right=of p6] (p7) {128};
    \node[obj, right=8mm of p7, draw=good, fill=good!10] (m) {median 128 \ok};
    \draw[arr] (q) -- (p1);
    \draw[arr] (p7) -- (m);
    \node[ann, above=1mm of p4] {7 independent passes};
  \end{tikzpicture}}
  \end{center}

  \vspace{2mm}
  \begin{itemize}
    \item Every pass returns a short rationale and a JSON answer array.
    \item A pass the parser cannot read is dropped, never counted as ``no
      objects'' --- it never happened in 5{,}392 validation calls.
    \item Aliases seen in training are canonicalised \emph{before} votes are
      counted, so equivalent surfaces reinforce rather than split a vote.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Step 3 --- Aggregate by cardinality, not by relation name}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \small
      \begin{tabular}{@{}lrl@{}}
        \toprule
        Relation & $n_r$ & Rule \\
        \midrule
        Area, capacity & 7 & median \\
        City of death & 20 & plurality vote, \emph{empty counts} \\
        Stock exchange & 10 & support $\ge 0.30$, empty vote \\
        Land borders & 3 & union \\
        Awards & 20 & support $\ge 0.05$ union \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{itemize}
        \item \textbf{One value:} agreement is the signal $\to$ median.
        \item \textbf{Zero or one:} silence is a vote $\to$ plurality.
        \item \textbf{Many:} a name seen once in twenty passes may still be
          right $\to$ low-threshold union.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{5mm}
  All thresholds were selected on \alert{training traces only}; validation
  gold is never touched before scoring.
\end{frame}

  \begin{columns}[T]

      \textcolor{other}{\textbf{Before}: gate, then generate}

      \vspace{2mm}

    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \textcolor{accent}{\textbf{After}: generate, then vote}

      \vspace{2mm}
      \begin{tikzpicture}[node distance=2mm]
        \node[sample] (a) {Paris};
        \node[sample, right=of a] (b) {Paris};
        \node[sampleempty, right=of b] (c) {$\varnothing$};
        \node[sample, right=of c] (d) {Paris};

        \draw[arr] (b.south east) -- (v);
      \end{tikzpicture}

      \vspace{2mm}
      \small Every pass may return an empty set; empties vote like any other

    \end{column}

  Also: few-shot demonstrations now mirror each relation's \emph{own} empty
  rate (2 of 5 for death cities, 1 of 5 for borders). A fixed 60\% empty mix
  had nudged the model toward silence on exactly the relations where silence
  is scored.
\end{frame}

\begin{frame}{Step 6 --- Enumerate big sets in slices}
  A single award may have 610 recipients; one undifferentiated prompt invites
  both truncation and omission.

  \vspace{2mm}
  \begin{center}
  \resizebox{0.95\textwidth}{!}{%
  \begin{tikzpicture}[node distance=2mm]
    \node[subj] (q) {award, \rel{awardWonBy}};
    \node[sample, right=8mm of q, minimum width=17mm] (w1) {era 1};
    \node[sample, right=of w1, minimum width=17mm] (w2) {era 2};
    \node[sample, right=of w2, minimum width=17mm] (w3) {era 3};
    \node[sample, right=of w3, minimum width=17mm] (w4) {era 4};
    \node[sample, right=of w4, minimum width=17mm] (w5) {era 5};
    \node[sample, right=of w5, minimum width=24mm, draw=accent, fill=accent!10] (w6) {by profession};
    \draw[arr] (q) -- (w1);
    \node[ann, above=1mm of w3.north east] {20 passes cycle through six foci; union with support $\ge 0.05$};
  \end{tikzpicture}}
  \end{center}

  \vspace{2mm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Token cap raised 2{,}048 $\to$ 4{,}096: a truncated JSON array
          parsed as \emph{zero} recipients, not a partial list.
        \item Mean predicted set size: 15 $\to$ 68 (gold mean 146).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item An earlier binary verifier cut correct recipients from 96 to 56,
          so we removed it: a closed-book verifier consults the same weights
          as the generator.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{What did not help, and why}

\begin{frame}{Takeaways}
  \begin{enumerate}
    \item \textbf{Sampling and aggregation are not where the score is.}
      Sample count, aggregation rule, abstention threshold, and even the choice
      between two eligible models all stay inside the confidence interval
      overall. The model alone moves \emph{individual} relations, in opposite
      directions.
    \item \textbf{Abstention is a prediction.} Let empty answers vote; never
      gate generation, and never ask an empty set what it is missing.
    \item \textbf{Report the ceiling with the score.} The share of rows whose
      correct value appears in \emph{any} sample tells you whether to spend
      effort on elicitation or on aggregation. For us the answer was
      elicitation.
  \end{enumerate}
